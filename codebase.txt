name: Deploy to GitHub Pages

on:
  # Автодеплой при пуше в main, если изменились файлы сайта или данных
  push:
    branches:
      - main
    paths:
      - 'docs/**'
      - 'data/**'
      - '.github/workflows/deploy.yml'

  # Возможность запустить вручную
  workflow_dispatch:

# Права для деплоя на Pages
permissions:
  contents: read
  pages: write
  id-token: write

# Только один деплой одновременно, старые прерываются
concurrency:
  group: "pages"
  cancel-in-progress: true

jobs:
  # =========================================================
  # 1. СБОРКА И ПРОВЕРКА
  # =========================================================
  build:
    name: Build site
    runs-on: ubuntu-latest
    timeout-minutes: 10

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Ensure docs/index.html exists
        run: |
          if [ ! -d "docs" ]; then
            echo "[*] Создаю папку docs/"
            mkdir -p docs
          fi

          if [ ! -f "docs/index.html" ]; then
            echo "[!] docs/index.html отсутствует — создаю заглушку"
            cat > docs/index.html <<'HTML'
          <!DOCTYPE html>
          <html lang="ru">
          <head>
            <meta charset="UTF-8">
            <meta name="viewport" content="width=device-width, initial-scale=1.0">
            <title>REMAININGCONNECTIONS · Остаться на связи</title>
            <style>
              body { font-family: system-ui, sans-serif; background:#0b0f19; color:#f9fafb; margin:0; min-height:100vh; display:flex; align-items:center; justify-content:center; text-align:center; padding:20px; }
              h1 { font-size: clamp(40px, 8vw, 96px); background:linear-gradient(90deg,#60a5fa,#a78bfa,#34d399); -webkit-background-clip:text; -webkit-text-fill-color:transparent; }
              p { color:#9ca3af; max-width:600px; }
              a { color:#60a5fa; }
            </style>
          </head>
          <body>
            <div>
              <h1>ОСТАТЬСЯ НА СВЯЗИ 🛜</h1>
              <p>Сайт задеплоен, но <code>docs/index.html</code> отсутствует.</p>
              <p><a href="https://t.me/REMAININGCONNECTIONS" target="_blank">@REMAININGCONNECTIONS</a></p>
            </div>
          </body>
          </html>
          HTML
          fi

          echo "[+] docs/index.html готов ($(wc -c < docs/index.html) bytes)"

      - name: Ensure data directory exists
        run: |
          mkdir -p data
          echo "[+] data/ готова"
          ls -lah data/

      - name: Setup Pages
        uses: actions/configure-pages@v5

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: ./docs

  # =========================================================
  # 2. ДЕПЛОЙ
  # =========================================================
  deploy:
    name: Deploy to Pages
    runs-on: ubuntu-latest
    needs: build
    timeout-minutes: 10

    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}

    steps:
      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4

      - name: Print success
        run: |
          echo "=================================================="
          echo " ✅ Сайт задеплоен!"
          echo "=================================================="
          echo " 🌐 URL: ${{ steps.deployment.outputs.page_url }}"
          echo " 🌍 Translate: https://translate.google.com/translate?sl=en&tl=ru&u=${{ steps.deployment.outputs.page_url }}"
          echo "=================================================="

          def write_json(path, payload):
              os.makedirs(os.path.dirname(path), exist_ok=True)
              with open(path, "w", encoding="utf-8") as f:
                  json.dump(payload, f, ensure_ascii=False, indent=2)
              print(f"[+] Сохранён JSON: {path}")

          def write_txt(path, entries):
              os.makedirs(os.path.dirname(path), exist_ok=True)
              with open(path, "w", encoding="utf-8") as f:
                  for entry in entries:
                      f.write(entry["full_query"] + "\n")
              print(f"[+] Сохранён TXT: {path}")

          def write_yaml(path, entries):
              os.makedirs(os.path.dirname(path), exist_ok=True)

              lines = [
                  "# Auto-generated search queries",
                  "# Do not edit manually if generated by CI.",
                  "queries:",
              ]

              for entry in entries:
                  lines.append(f"  - slug: {json.dumps(entry['slug'], ensure_ascii=False)}")
                  lines.append(f"    query: {json.dumps(entry['query'], ensure_ascii=False)}")
                  lines.append(f"    full_query: {json.dumps(entry['full_query'], ensure_ascii=False)}")
                  lines.append(f"    category: {json.dumps(entry['category'], ensure_ascii=False)}")
                  lines.append(f"    type: {json.dumps(entry['type'], ensure_ascii=False)}")
                  lines.append(f"    priority: {entry['priority']}")
                  lines.append(f"    reason: {json.dumps(entry['reason'], ensure_ascii=False)}")

              with open(path, "w", encoding="utf-8") as f:
                  f.write("\n".join(lines) + "\n")

              print(f"[+] Сохранён YAML: {path}")

          def write_matrix_json(path, entries):

                          "score": 1.0,

          import os

          candidates_dir = "candidates"

          seen = set()
          total = 0
          written = 0

          os.makedirs(raw_dir, exist_ok=True)

          patterns = [
              os.path.join(candidates_dir, "**", "*.jsonl"),
              os.path.join(raw_dir, "**", "*.jsonl"),
          ]

          files = []
          for pattern in patterns:
              files.extend(glob.glob(pattern, recursive=True))

          files = [f for f in files if os.path.basename(f) != "all_candidates.jsonl"]

          with open(out_path, "w", encoding="utf-8") as out:

                                  continue

                              try:
                                  obj = json.loads(line)

                                  continue

                              source = obj.get("source")

                              if not repo or not source:
                                  continue

                              key = f"{source}::{repo.lower()}::{path.lower()}"

                              if key in seen:
                                  continue

                              seen.add(key)

                              normalized = {
                                  "source": source,
                                  "kind": obj.get("kind", "repo"),
                                  "repo": repo,
                                  "url": obj.get("url"),
                                  "path": path or None,
                                  "score": obj.get("score", 0),

                              written += 1
                  except Exception as e:
                      print(f"[!] Ошибка чтения {file_path}: {e}")

          print(f"[+] Уникальных: {written}")
          PY

  probe-candidates:
    name: 7 · Проверка кандидатов
    runs-on: ubuntu-latest
    needs: collect-candidates
    if: always()
    timeout-minutes: 45

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: ${{ env.PYTHON_VERSION }}

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install requests pyyaml

      - name: Create directories
        run: |
          mkdir -p candidates
          mkdir -p raw
          mkdir -p "${OUTPUT_DIR}"

      - name: Download candidates
        uses: actions/download-artifact@v4
        continue-on-error: true
        with:
          name: all-candidates
          path: raw

      - name: Probe repositories
        env:
          GITHUB_TOKEN: ${{ github.token }}
          GITVERSE_BASE: "https://gitverse.ru"
          MAX_PROBE: ${{ env.MAX_PROBE }}
          USER_AGENT: ${{ env.USER_AGENT }}
        run: |
          python <<'PY'
          import os
          import re
          import json
          import time
          import datetime
          import requests
          import yaml

          GITHUB_API = "https://api.github.com"
          GITVERSE_BASE = os.getenv("GITVERSE_BASE", "https://gitverse.ru")
          TOKEN = os.getenv("GITHUB_TOKEN", "")
          USER_AGENT = os.getenv("USER_AGENT", "REMAININGCONNECTIONS/1.0")
          MAX_PROBE = int(os.getenv("MAX_PROBE", "220"))
          OUTPUT_DIR = os.getenv("OUTPUT_DIR", "data")

          HEADERS = {
              "Accept": "application/vnd.github+json",
              "User-Agent": USER_AGENT,
          }

          if TOKEN:
              HEADERS["Authorization"] = f"Bearer {TOKEN}"

          COMMON_PATHS = [
              "sub.txt", "subscribe.txt", "subscription.txt", "subscriptions.txt",
              "v2ray.txt", "xray.txt", "nodes.txt", "proxy.txt", "proxies.txt",
              "list.txt", "clash.yaml", "clash.yml", "config.yaml", "config.yml",
              "sub.yaml", "sub.yml", "subscription.yaml", "subscription.yml",
              "proxies.yaml", "proxies.yml", "mihomo.yaml", "mihomo.yml",
              "sing-box.json", "singbox.json", "wireguard.conf", "wg.conf",
          ]

          PROTO_RE = re.compile(
              r"^(vless|vmess|ss|ssr|trojan|trojan-go|hysteria|hysteria2|hy2|tuic|wireguard|socks5|http|https|tg|mtproto)://",
              re.IGNORECASE,
          )

          BS_MARKERS = [
              "whitelist", "white_list", "white list", "белый список", "белые списки",
              "белый-список", "bs: true", "bs=true", '"bs": true', "'bs': true",
              "bypass_list", "bypass-list",
          ]

          session = requests.Session()
          session.headers.update(HEADERS)

          def now_utc():
              return datetime.datetime.now(datetime.timezone.utc)

          def parse_date(value):
              if not value:
                  return None
              try:
                  value = value.replace("Z", "+00:00")
                  dt = datetime.datetime.fromisoformat(value)
                  if dt.tzinfo is None:
                      dt = dt.replace(tzinfo=datetime.timezone.utc)
                  return dt
              except Exception:
                  return None

          def minutes_ago(dt):
              if not dt:
                  return None
              delta = now_utc() - dt
              return int(delta.total_seconds() / 60)

          def fetch_json(url, params=None):
              for attempt in range(3):
                  try:
                      r = session.get(url, params=params, timeout=25)
                      if r.status_code == 200:
                          return r.json()
                      if r.status_code in (403, 429):
                          retry_after = r.headers.get("Retry-After")
                          wait = int(retry_after) if retry_after and retry_after.isdigit() else 6
                          print(f"[!] Rate limit: {r.status_code}, жду {wait}с")
                          time.sleep(wait)
                          continue
                      if 500 <= r.status_code < 600:
                          time.sleep(2 + attempt)
                          continue
                      return None
                  except Exception as e:
                      print(f"[!] Ошибка запроса {url}: {e}")
                      time.sleep(2 + attempt)
              return None

          def fetch_text(url):
              for attempt in range(2):
                  try:
                      r = session.get(url, timeout=25)
                      if r.status_code == 200 and r.text and len(r.text.strip()) > 0:
                          return r.text
                      return None
                  except Exception as e:
                      print(f"[!] Ошибка загрузки {url}: {e}")
                      time.sleep(1 + attempt)
              return None

          def check_bs(text):
              if not text:
                  return False
              low = text.lower()
              return any(marker in low for marker in BS_MARKERS)

          def count_text_configs(text):
              if not text:
                  return 0
              lines = [l.strip() for l in text.splitlines() if l.strip() and not l.strip().startswith("#") and not l.strip().startswith("//")]
              if not lines:
                  return 0
              proto_lines = [x for x in lines if PROTO_RE.match(x)]
              if len(proto_lines) >= max(1, int(len(lines) * 0.25)):
                  return len(proto_lines)
              return len(lines)

          def count_yaml_configs(text):
              if not text:
                  return 0
              try:
                  doc = yaml.safe_load(text)
                  if isinstance(doc, dict):
                      if isinstance(doc.get("proxies"), list):
                          return len(doc["proxies"])
                      if isinstance(doc.get("proxy-providers"), dict):
                          return len(doc["proxy-providers"])
                      if isinstance(doc.get("nodes"), list):
                          return len(doc["nodes"])
                  if isinstance(doc, list):
                      return len(doc)
              except Exception:
                  pass
              return count_text_configs(text)

          def count_by_path(text, path):
              p = (path or "").lower()
              if p.endswith((".yaml", ".yml")):
                  return count_yaml_configs(text)
              return count_text_configs(text)

          def make_name(repo):
              return repo.split("/")[-1] if repo else "Unknown"

          def probe_github_repo(repo, candidate_path=None):
              print(f"[*] Проверяю: {repo}")

              api_repo = f"{GITHUB_API}/repos/{repo}"
              repo_data = fetch_json(api_repo)

              if not repo_data:
                  print(f"[!] Не могу получить {repo}")
                  return None

              default_branch = repo_data.get("default_branch", "main")
              html_url = repo_data.get("html_url", f"https://github.com/{repo}")

              commits = fetch_json(f"{api_repo}/commits", params={"per_page": 1})
              updated_mins = None
              last_commit_message = None

              if isinstance(commits, list) and commits:
                  commit = commits[0]
                  commit_data = commit.get("commit", {})
                  committer = commit_data.get("committer", {})
                  date_str = committer.get("date") or commit_data.get("author", {}).get("date")
                  updated_mins = minutes_ago(parse_date(date_str))
                  last_commit_message = commit_data.get("message")

              paths = []
              if candidate_path:
                  paths.append(candidate_path)
              paths.extend(COMMON_PATHS)

              seen = set()
              unique_paths = []
              for p in paths:
                  if p and p not in seen:
                      seen.add(p)
                      unique_paths.append(p)

              found_url = None
              found_path = None
              configs_count = 0
              has_bs = False
              content_sample = ""

              for path in unique_paths[:18]:
                  raw_url = f"https://raw.githubusercontent.com/{repo}/{default_branch}/{path}"
                  text = fetch_text(raw_url)

                  if not text:
                      continue

                  cnt = count_by_path(text, path)
                  bs = check_bs(text)

                  if cnt > 0:
                      found_url = raw_url
                      found_path = path
                      configs_count = cnt
                      has_bs = bs
                      content_sample = text[:800]
                      break

                  if not found_url:
                      found_url = raw_url
                      found_path = path
                      content_sample = text[:800]
                      has_bs = bs

                  time.sleep(1 + (time.time() % 2))

              status = "active"
              if configs_count <= 0:
                  status = "unknown"
              elif updated_mins is not None and updated_mins > 60 * 24 * 14:
                  status = "stale"

              return {
                  "name": make_name(repo),
                  "source": "github",
                  "repo": repo,
                  "url": html_url,
                  "subscription_url": found_url,
                  "config_path": found_path,
                  "updated_mins_ago": updated_mins,
                  "configs_count": configs_count,
                  "has_bs": has_bs,
                  "status": status,
                  "last_commit_message": last_commit_message,
                  "content_sample": content_sample,
              }

          def probe_gitverse_repo(repo, candidate_path=None):
              print(f"[*] Проверяю Gitverse: {repo}")

              api_repo = f"{GITVERSE_BASE}/api/v1/repos/{repo}"
              repo_data = fetch_json(api_repo)

              default_branch = "main"
              html_url = f"{GITVERSE_BASE}/{repo}"

              if repo_data:
                  default_branch = repo_data.get("default_branch", "main") or "main"
                  html_url = repo_data.get("html_url") or html_url

              commits = fetch_json(f"{api_repo}/commits", params={"limit": 1})
              updated_mins = None
              last_commit_message = None

              if isinstance(commits, list) and commits:
                  commit = commits[0]
                  commit_obj = commit.get("commit", {})
                  committer = commit_obj.get("committer", {})
                  date_str = committer.get("date") or commit_obj.get("author", {}).get("date")
                  updated_mins = minutes_ago(parse_date(date_str))
                  last_commit_message = commit_obj.get("message")

              paths = []
              if candidate_path:
                  paths.append(candidate_path)
              paths.extend(COMMON_PATHS)

              seen = set()
              unique_paths = []
              for p in paths:
                  if p and p not in seen:
                      seen.add(p)
                      unique_paths.append(p)

              branches = [default_branch, "main", "master"]
              found_url = None
              found_path = None
              configs_count = 0
              has_bs = False
              content_sample = ""

              for branch in branches:
                  if found_url and configs_count > 0:
                      break

                  for path in unique_paths[:12]:
                      raw_candidates = [
                          f"{GITVERSE_BASE}/{repo}/raw/branch/{branch}/{path}",
                          f"{GITVERSE_BASE}/{repo}/raw/{branch}/{path}",
                          f"{GITVERSE_BASE}/{repo}/raw/{path}",
                      ]

                      for raw_url in raw_candidates:
                          text = fetch_text(raw_url)
                          if not text:
                              continue

                          cnt = count_by_path(text, path)
                          bs = check_bs(text)

                          if cnt > 0:
                              found_url = raw_url
                              found_path = path
                              configs_count = cnt
                              has_bs = bs
                              content_sample = text[:800]
                              break

                          if not found_url:
                              found_url = raw_url
                              found_path = path
                              content_sample = text[:800]
                              has_bs = bs

                      if found_url and configs_count > 0:
                          break

                      time.sleep(1 + (time.time() % 2))

              status = "active"
              if configs_count <= 0:
                  status = "unknown"
              elif updated_mins is not None and updated_mins > 60 * 24 * 14:
                  status = "stale"

              return {
                  "name": make_name(repo),
                  "source": "gitverse",
                  "repo": repo,
                  "url": html_url,
                  "subscription_url": found_url,
                  "config_path": found_path,
                  "updated_mins_ago": updated_mins,
                  "configs_count": configs_count,
                  "has_bs": has_bs,
                  "status": status,
                  "last_commit_message": last_commit_message,
                  "content_sample": content_sample,
              }

          def load_candidates():
              candidates = []
              path = "raw/all_candidates.jsonl"

              if not os.path.exists(path):
                  print(f"[!] Файл не найден: {path}")
                  return candidates

              with open(path, "r", encoding="utf-8") as f:
                  for line in f:
                      line = line.strip()
                      if not line:
                          continue
                      try:
                          candidates.append(json.loads(line))
                      except Exception:
                          continue

              return candidates

          candidates = load_candidates()
          print(f"[*] Загружено кандидатов: {len(candidates)}")

          candidates.sort(
              key=lambda x: (
                  x.get("source") != "github",
                  x.get("path") is None,
                  -(x.get("score") or 0),
              )
          )

          results = []
          probed_repos = set()

          for candidate in candidates:
              if len(results) >= MAX_PROBE:
                  print(f"[*] Достигнут лимит: {MAX_PROBE}")
                  break

              repo = candidate.get("repo")
              source = candidate.get("source")
              path = candidate.get("path")

              if not repo or not source:
                  continue

              repo_key = f"{source}::{repo.lower()}"
              if repo_key in probed_repos:
                  continue

              probed_repos.add(repo_key)

              try:
                  if source == "github":
                      item = probe_github_repo(repo, path)
                  elif source == "gitverse":
                      item = probe_gitverse_repo(repo, path)
                  else:
                      item = None

                  if item:
                      results.append(item)

              except Exception as e:
                  print(f"[!] Ошибка проверки {repo_key}: {e}")

              time.sleep(1 + (time.time() % 3))

          output = {
              "generated_at": now_utc().isoformat().replace("+00:00", "Z"),
              "generator": "subscription-discovery.yml",
              "candidates_found": len(candidates),
              "probed_count": len(results),
              "subscriptions": results,
          }

          os.makedirs(OUTPUT_DIR, exist_ok=True)
          out_file = os.path.join(OUTPUT_DIR, "subscriptions_found.json")

          with open(out_file, "w", encoding="utf-8") as f:
              json.dump(output, f, ensure_ascii=False, indent=2)

          print(f"[+] Сохранено {len(results)} подписок в {out_file}")
          PY

          retention-days: 14

    steps:

      - name: Extract TG proxies
        env:
          GITHUB_TOKEN: ${{ github.token }}
          GITVERSE_BASE: "https://gitverse.ru"
          MAX_FETCH_CANDIDATES: ${{ env.MAX_FETCH_CANDIDATES }}
          USER_AGENT: ${{ env.USER_AGENT }}
        run: |
          python <<'PY'
          import os
          import re
          import json
          import time
          import random
          import requests
          from urllib.parse import urlparse, parse_qs

          GITHUB_API = "https://api.github.com"
          GITVERSE_BASE = os.getenv("GITVERSE_BASE", "https://gitverse.ru")
          TOKEN = os.getenv("GITHUB_TOKEN", "")
          USER_AGENT = os.getenv("USER_AGENT", "REMAININGCONNECTIONS/1.0")
          MAX_FETCH_CANDIDATES = int(os.getenv("MAX_FETCH_CANDIDATES", "300"))

          HEADERS = {
              "Accept": "application/vnd.github+json",
              "User-Agent": USER_AGENT,
          }

          if TOKEN:
              HEADERS["Authorization"] = f"Bearer {TOKEN}"

          session = requests.Session()
          session.headers.update(HEADERS)

          TG_PROXY_RE = re.compile(
              r"(?:tg://proxy|https?://t\.me/proxy)\?[^\s\)\]\"'<>]+",
              re.IGNORECASE,
          )

          COMMON_TG_FILES = [
              "mtproxy.txt", "mtproxy-list.txt", "mtproto.txt",
              "tg-proxy.txt", "tg-proxies.txt", "telegram-proxy.txt",
              "telegram-proxies.txt", "proxy.txt", "proxies.txt",
              "proxy-list.txt", "list.txt", "tg.txt", "tg.md",
              "proxy.md", "proxies.md", "mtproxy.md", "mtproto.md",
              "README.md", "README.RU.md", "readme.md",
              "free-tg-proxy.txt", "tg-proxy.json", "mtproxy.json",
              "mtproto.json", "proxy.yaml", "proxies.yaml",
          ]

          branch_cache = {}
          proxies = {}

          def sleep_random(min_seconds=1, max_seconds=3):
              time.sleep(min_seconds + random.random() * (max_seconds - min_seconds))

          def fetch_json(url, params=None):
              for attempt in range(3):
                  try:
                      r = session.get(url, params=params, timeout=25)
                      if r.status_code == 200:
                          return r.json()
                      if r.status_code in (403, 429):
                          retry_after = r.headers.get("Retry-After")
                          wait = int(retry_after) if retry_after and retry_after.isdigit() else 6
                          print(f"[!] Rate limit: {r.status_code}, жду {wait}с")
                          time.sleep(wait)
                          continue
                      if 500 <= r.status_code < 600:
                          time.sleep(2 + attempt)
                          continue
                      return None
                  except Exception as e:
                      print(f"[!] Ошибка запроса {url}: {e}")
                      time.sleep(2 + attempt)
              return None

          def fetch_text(url):
              try:
                  r = session.get(url, timeout=25)
                  if r.status_code == 200 and r.text and len(r.text.strip()) > 0:
                      return r.text
                  return None
              except Exception as e:
                  print(f"[!] Ошибка загрузки {url}: {e}")
                  return None

          def get_github_default_branch(repo):
              if repo in branch_cache:
                  return branch_cache[repo]

              data = fetch_json(f"{GITHUB_API}/repos/{repo}")
              branch = None

              if data:
                  branch = data.get("default_branch")

              if not branch:
                  branch = "main"

              branch_cache[repo] = branch
              return branch

          def get_gitverse_default_branch(repo):
              if repo in branch_cache:
                  return branch_cache[repo]

              data = fetch_json(f"{GITVERSE_BASE}/api/v1/repos/{repo}")
              branch = None

              if data:
                  branch = data.get("default_branch")

              if not branch:
                  branch = "main"

              branch_cache[repo] = branch
              return branch

          def parse_tg_url(url):
              try:
                  parsed = urlparse(url)
                  params = parse_qs(parsed.query)

                  server = params.get("server", [None])[0]
                  port = params.get("port", [None])[0]
                  secret = params.get("secret", [None])[0]

                  if not server or not port or not secret:
                      return None

                  port = int(port)

                  if port <= 0 or port > 65535:
                      return None

                  if not re.fullmatch(r"[a-zA-Z0-9.-]{3,255}", server):
                      return None

                  secret = secret.strip()

                  if not re.fullmatch(r"[0-9a-fA-F]{32,128}", secret):
                      return None

                  return {
                      "server": server,
                      "port": port,
                      "secret": secret,
                  }
              except Exception:
                  return None

          def extract_from_text(text, source, repo, path, candidate_url):
              if not text:
                  return 0

              found = 0
              matches = TG_PROXY_RE.findall(text)

              for match in matches:
                  parsed = parse_tg_url(match)

                  if not parsed:
                      continue

                  key = f"{parsed['server']}:{parsed['port']}:{parsed['secret']}"

                  tg_url = (
                      f"tg://proxy?server={parsed['server']}"
                      f"&port={parsed['port']}"
                      f"&secret={parsed['secret']}"
                  )

                  tme_url = (
                      f"https://t.me/proxy?server={parsed['server']}"
                      f"&port={parsed['port']}"
                      f"&secret={parsed['secret']}"
                  )

                  source_ref = {
                      "source": source,
                      "repo": repo,
                      "path": path,
                      "url": candidate_url,
                  }

                  if key not in proxies:
                      proxies[key] = {
                          "server": parsed["server"],
                          "port": parsed["port"],
                          "secret": parsed["secret"],
                          "protocol": "mtproto",
                          "tg_url": tg_url,
                          "tme_url": tme_url,
                          "sources": [source_ref],
                      }
                  else:
                      if len(proxies[key]["sources"]) < 8:
                          proxies[key]["sources"].append(source_ref)

                  found += 1

              return found

          def fetch_github_paths(repo, path=None):
              branch = get_github_default_branch(repo)
              paths = []

              if path:
                  paths.append(path)
              else:
                  paths.extend(COMMON_TG_FILES)

              for p in paths[:20]:
                  raw_url = f"https://raw.githubusercontent.com/{repo}/{branch}/{p}"
                  text = fetch_text(raw_url)

                  if text:
                      yield p, raw_url, text

                  sleep_random(0.5, 1.5)

          def fetch_gitverse_paths(repo, path=None):
              branch = get_gitverse_default_branch(repo)
              paths = []

              if path:
                  paths.append(path)
              else:
                  paths.extend(COMMON_TG_FILES)

              branches = [branch, "main", "master"]

              for b in branches:
                  for p in paths[:12]:
                      raw_candidates = [
                          f"{GITVERSE_BASE}/{repo}/raw/branch/{b}/{p}",
                          f"{GITVERSE_BASE}/{repo}/raw/{b}/{p}",
                          f"{GITVERSE_BASE}/{repo}/raw/{p}",
                      ]

                      for raw_url in raw_candidates:
                          text = fetch_text(raw_url)

                          if text:
                              yield p, raw_url, text
                              break

                      sleep_random(0.5, 1.5)

          candidates = []
          candidate_file = "raw/tg_candidates.jsonl"

          if os.path.exists(candidate_file):
              with open(candidate_file, "r", encoding="utf-8") as f:
                  for line in f:
                      line = line.strip()
                      if not line:
                          continue

                      try:
                          candidates.append(json.loads(line))
                      except Exception:
                          continue
          else:
              print("[!] Файл кандидатов не найден")

          print(f"[*] Загружено кандидатов: {len(candidates)}")

          candidates.sort(
              key=lambda x: (
                  x.get("source") != "github",
                  x.get("path") is None,
                  -(x.get("score") or 0),
              )
          )

          processed = 0
          total_found = 0

          for candidate in candidates:
              if processed >= MAX_FETCH_CANDIDATES:
                  print(f"[*] Достигнут лимит: {MAX_FETCH_CANDIDATES}")
                  break

              source = candidate.get("source")
              repo = candidate.get("repo")
              path = candidate.get("path")
              candidate_url = candidate.get("url")

              if not source or not repo:
                  continue

              processed += 1
              print(f"[*] Обрабатываю {source}:{repo}:{path or '<без пути>'}")

              try:
                  if source == "github":
                      iterator = fetch_github_paths(repo, path)
                  elif source == "gitverse":
                      iterator = fetch_gitverse_paths(repo, path)
                  else:
                      iterator = []

                  for file_path, raw_url, text in iterator:
                      found = extract_from_text(
                          text=text,
                          source=source,
                          repo=repo,
                          path=file_path,
                          candidate_url=raw_url,
                      )

                      total_found += found

                      if found:
                          print(f"[+] Найдено {found} прокси в {raw_url}")

              except Exception as e:
                  print(f"[!] Ошибка обработки {repo}: {e}")

              sleep_random(1, 2)

          os.makedirs("extracted", exist_ok=True)

          out_jsonl = "extracted/tg_proxies_extracted.jsonl"
          out_json = "extracted/tg_proxies_extracted.json"

          proxy_list = list(proxies.values())

          with open(out_jsonl, "w", encoding="utf-8") as f:
              for item in proxy_list:
                  f.write(json.dumps(item, ensure_ascii=False) + "\n")

          payload = {
              "extracted_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
              "generator": "tg-proxy-discovery.yml",
              "processed_candidates": processed,
              "unique_proxies": len(proxy_list),
              "raw_mentions_found": total_found,
              "proxies": proxy_list,
          }

          with open(out_json, "w", encoding="utf-8") as f:
              json.dump(payload, f, ensure_ascii=False, indent=2)

          print(f"[*] Обработано кандидатов: {processed}")
          print(f"[*] Всего упоминаний: {total_found}")
          print(f"[+] Уникальных прокси: {len(proxy_list)}")
          PY

      - name: Upload extracted proxies
        uses: actions/upload-artifact@v4
        with:
          name: tg-proxies-extracted
          path: extracted/tg_proxies_extracted.*
          if-no-files-found: warn
          retention-days: 14

          def classify_secret(secret):

                  return "unknown"

              s = secret.lower()

              if s.startswith("dd"):
                  return "fake_tls_or_padding"
              if s.startswith("ee"):
                  return "media_or_special"
              if len(s) == 32 or len(s) == 64:
                  return "hex"

              return "other"

          async def check_one(proxy, semaphore):
              async with semaphore:
                  server = proxy.get("server")
                  port = proxy.get("port")
                  secret = proxy.get("secret")

                  result = {
                      "id": None,
                      "host": server,
                      "server": server,
                      "port": port,
                      "secret": secret,
                      "protocol": "MTProto",
                      "tg_url": proxy.get("tg_url"),
                      "tme_url": proxy.get("tme_url"),
                      "sources": proxy.get("sources", []),
                      "secret_type": classify_secret(secret),
                      "checked_at": now_iso(),
                      "dns_ok": False,
                      "tcp_open": False,
                      "latency_ms": None,
                      "ping": None,
                      "status": "offline",
                      "country": None,
                      "errors": [],
                  }

                  if not server or not port:
                      result["errors"].append("invalid_server_or_port")
                      return result

                  try:
                      port = int(port)
                      result["port"] = port
                  except Exception:
                      result["errors"].append("invalid_port")
                      return result

                  # DNS check
                  try:
                      infos = await asyncio.get_running_loop().getaddrinfo(
                          server,
                          port,
                          proto=socket.IPPROTO_TCP,
                      )

                      if not infos:
                          result["errors"].append("dns_no_records")
                          return result

                      result["dns_ok"] = True
                  except Exception as e:
                      result["errors"].append(f"dns_error:{e.__class__.__name__}")
                      return result

                  # TCP check with retries
                  retries = 2

                  for attempt in range(retries):
                      start = time.perf_counter()

                      try:
                          reader, writer = await asyncio.wait_for(
                              asyncio.open_connection(server, port),
                              timeout=CHECK_TIMEOUT,
                          )

                          latency_ms = round((time.perf_counter() - start) * 1000, 2)

                          result["tcp_open"] = True
                          result["latency_ms"] = latency_ms
                          result["ping"] = latency_ms
                          result["status"] = "online"

                          try:
                              writer.close()
                              await writer.wait_closed()
                          except Exception:
                              pass

                          break

                      except asyncio.TimeoutError:
                          result["errors"].append(f"tcp_timeout_attempt_{attempt + 1}")
                      except Exception as e:
                          result["errors"].append(f"tcp_error:{e.__class__.__name__}")

                      await asyncio.sleep(0.25)

                  return result

          async def main():
              proxies = load_proxies()

              if not proxies:
                  payload = {
                      "generated_at": now_iso(),
                      "generator": "tg-proxy-discovery.yml",
                      "checked_count": 0,
                      "online_count": 0,
                      "offline_count": 0,
                      "dns_fail_count": 0,
                      "avg_latency_ms": None,
                      "proxies": [],
                  }

                  os.makedirs("checked", exist_ok=True)
                  os.makedirs(os.path.dirname(FINAL_JSON), exist_ok=True)

                  with open(OUTPUT_JSON, "w", encoding="utf-8") as f:
                      json.dump(payload, f, ensure_ascii=False, indent=2)

                  with open(FINAL_JSON, "w", encoding="utf-8") as f:
                      json.dump(payload, f, ensure_ascii=False, indent=2)

                  print("[!] Нет прокси для проверки. Созданы пустые файлы.")
                  return

              # Сортируем: у кого больше источников, те приоритетнее
              proxies.sort(
                  key=lambda x: len(x.get("sources", [])),
                  reverse=True,
              )

              proxies = proxies[:MAX_CHECK_PROXIES]

              print(f"[*] Проверяю {len(proxies)} прокси")
              print(f"[*] Таймаут: {CHECK_TIMEOUT}с")
              print(f"[*] Одновременно: {CONCURRENCY_LIMIT}")

              semaphore = asyncio.Semaphore(CONCURRENCY_LIMIT)

              tasks = [
                  check_one(proxy, semaphore)
                  for proxy in proxies
              ]

              results = await asyncio.gather(*tasks, return_exceptions=False)

              # Проставляем ID после проверки
              for idx, item in enumerate(results, start=1):
                  item["id"] = idx

              online = [x for x in results if x["status"] == "online"]
              offline = [x for x in results if x["status"] != "online"]
              dns_fail = [x for x in results if not x["dns_ok"]]

              avg_latency = None
              if online:
                  latencies = [x["latency_ms"] for x in online if x.get("latency_ms") is not None]
                  if latencies:
                      avg_latency = round(sum(latencies) / len(latencies), 2)

              # Сортировка итога: онлайн сверху, по пингу
              results.sort(
                  key=lambda x: (
                      x["status"] != "online",
                      x["latency_ms"] if x.get("latency_ms") is not None else 999999,
                      -(len(x.get("sources", []))),
                  )
              )

              payload = {
                  "generated_at": now_iso(),
                  "generator": "tg-proxy-discovery.yml",
                  "telegram_channel": "https://t.me/REMAININGCONNECTIONS",
                  "checked_count": len(results),
                  "online_count": len(online),
                  "offline_count": len(offline),
                  "dns_fail_count": len(dns_fail),
                  "avg_latency_ms": avg_latency,
                  "check_timeout_seconds": CHECK_TIMEOUT,
                  "concurrency_limit": CONCURRENCY_LIMIT,
                  "proxies": results,
              }

              os.makedirs("checked", exist_ok=True)
              os.makedirs(os.path.dirname(FINAL_JSON), exist_ok=True)

              with open(OUTPUT_JSON, "w", encoding="utf-8") as f:
                  json.dump(payload, f, ensure_ascii=False, indent=2)

              with open(FINAL_JSON, "w", encoding="utf-8") as f:
                  json.dump(payload, f, ensure_ascii=False, indent=2)

              print("==================================================")
              print(" Проверка ТГ прокси завершена")
              print("==================================================")
              print(f"Проверено: {len(results)}")
              print(f"Онлайн: {len(online)}")
              print(f"Оффлайн: {len(offline)}")
              print(f"DNS ошибки: {len(dns_fail)}")
              print(f"Средний пинг: {avg_latency}")
              print(f"Сохранено: {OUTPUT_JSON}")
              print(f"Сохранено: {FINAL_JSON}")
              print("==================================================")

          asyncio.run(main())
          PY

              if '.git' not in root

        run: |
          python <<'PY'
          import os
          import json
          import glob

          files = glob.glob("candidates/**/*.jsonl", recursive=True)
          files += glob.glob("raw/**/*.jsonl", recursive=True)

          out_path = "raw/utils_candidates.jsonl"

          seen = {}
          total = 0

          for file_path in sorted(files):
              try:
                  with open(file_path, "r", encoding="utf-8") as f:
                      for line in f:
                          line = line.strip()
                          if not line:
                              continue

                          total += 1

                          try:
                              obj = json.loads(line)
                          except Exception:
                              continue

                          repo = obj.get("repo")
                          if not repo:
                              continue

                          if repo not in seen:
                              seen[repo] = obj
                          else:
                              old = seen[repo]

                              old_slugs = old.get("query_slugs", [])
                              if old.get("query_slug") and old["query_slug"] not in old_slugs:
                                  old_slugs.append(old["query_slug"])

                              new_slug = obj.get("query_slug")
                              if new_slug and new_slug not in old_slugs:
                                  old_slugs.append(new_slug)

                              old["query_slugs"] = old_slugs

                              if obj.get("stars", 0) > old.get("stars", 0):
                                  old["stars"] = obj.get("stars", 0)

                              if len(obj.get("description") or "") > len(old.get("description") or ""):
                                  old["description"] = obj.get("description")

                              old_topics = set(old.get("topics") or [])
                              new_topics = set(obj.get("topics") or [])
                              old["topics"] = list(old_topics | new_topics)

              except Exception as e:
                  print(f"[!] Ошибка чтения {file_path}: {e}")

          candidates = list(seen.values())
          candidates.sort(key=lambda x: x.get("stars", 0), reverse=True)

          os.makedirs("raw", exist_ok=True)

          with open(out_path, "w", encoding="utf-8") as out:
              for item in candidates:
                  out.write(json.dumps(item, ensure_ascii=False) + "\n")

          print(f"[*] Всего кандидатов: {total}")
          print(f"[+] Уникальных: {len(candidates)}")
          PY

      - name: Upload candidates
        uses: actions/upload-artifact@v4
        with:
          name: utils-candidates
          path: raw/utils_candidates.jsonl
          if-no-files-found: warn
          retention-days: 7

  # =========================================================
  # 5. ПРОВЕРКА УТИЛИТ
  # =========================================================
  probe-utilities:
    name: 5 · Проверка утилит
    runs-on: ubuntu-latest
    needs: collect-candidates
    if: always()
    timeout-minutes: 70

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: ${{ env.PYTHON_VERSION }}

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install requests

      - name: Create directories
        run: |
          mkdir -p raw
          mkdir -p probed
          mkdir -p "${OUTPUT_DIR}"

      - name: Download candidates
        uses: actions/download-artifact@v4
        continue-on-error: true
        with:
          name: utils-candidates
          path: raw

      - name: Probe utilities
        env:
          GITHUB_TOKEN: ${{ github.token }}
          MAX_PROBE: ${{ env.MAX_PROBE }}
          INCLUDE_README: ${{ env.INCLUDE_README }}
          USER_AGENT: ${{ env.USER_AGENT }}
          MIN_STARS: ${{ env.MIN_STARS }}
        run: |
          python <<'PY'
          import os
          import re
          import json
          import time
          import random
          import datetime
          import requests

          GITHUB_API = "https://api.github.com"
          TOKEN = os.getenv("GITHUB_TOKEN", "")
          USER_AGENT = os.getenv("USER_AGENT", "REMAININGCONNECTIONS/1.0")
          MAX_PROBE = int(os.getenv("MAX_PROBE", "300"))
          INCLUDE_README = os.getenv("INCLUDE_README", "1") == "1"
          MIN_STARS = int(os.getenv("MIN_STARS", "0"))
          UTILS_OUTPUT = os.getenv("OUTPUT_FILE", "data/utils_found.json")

          HEADERS = {
              "Accept": "application/vnd.github+json",
              "User-Agent": USER_AGENT,
          }

          if TOKEN:
              HEADERS["Authorization"] = f"Bearer {TOKEN}"

          session = requests.Session()
          session.headers.update(HEADERS)

          ANDROID_KEYS = [
              "android", ".apk", ".aab", "apk", "play google",
              "f-droid", "fdroid", "android app", "android client",
          ]

          IOS_KEYS = [
              "ios", "iphone", "ipad", ".ipa", "ipa",
              "testflight", "app store", "ios app", "ios client",
          ]

          WINDOWS_KEYS = [
              "windows", "win64", "win32", "win-arm", ".exe",
              ".msi", ".msix", ".appx", "portable.exe", "windows app",
          ]

          LINUX_KEYS = [
              "linux", "ubuntu", "debian", "fedora", "archlinux",
              "arch linux", "appimage", ".deb", ".rpm", ".snap",
              ".flatpak", ".pkg.tar.zst", ".tar.gz", ".tar.xz",
              ".tar.zst", ".tar.bz2", "linux app",
          ]

          def now_iso():
              return (
                  datetime.datetime.now(datetime.timezone.utc)
                  .isoformat()
                  .replace("+00:00", "Z")
              )

          def parse_date(value):
              if not value:
                  return None
              try:
                  value = value.replace("Z", "+00:00")
                  dt = datetime.datetime.fromisoformat(value)
                  if dt.tzinfo is None:
                      dt = dt.replace(tzinfo=datetime.timezone.utc)
                  return dt
              except Exception:
                  return None

          def days_since(dt):
              if not dt:
                  return None
              return int((datetime.datetime.now(datetime.timezone.utc) - dt).total_seconds() / 86400)

          def fetch_json(url, params=None):
              for attempt in range(3):
                  try:
                      r = session.get(url, params=params, timeout=25)
                      if r.status_code == 200:
                          return r.json()
                      if r.status_code == 404:
                          return None
                      if r.status_code in (403, 429):
                          retry_after = r.headers.get("Retry-After")
                          wait = int(retry_after) if retry_after and retry_after.isdigit() else 6
                          print(f"[!] Rate limit: {r.status_code}, жду {wait}с")
                          time.sleep(wait)
                          continue
                      if 500 <= r.status_code < 600:
                          time.sleep(2 + attempt)
                          continue
                      return None
                  except Exception as e:
                      print(f"[!] Ошибка запроса {url}: {e}")
                      time.sleep(2 + attempt)
              return None

          def fetch_text(url, accept=None):
              try:
                  headers = {}
                  if accept:
                      headers["Accept"] = accept
                  r = session.get(url, headers=headers, timeout=25)
                  if r.status_code == 200 and r.text:
                      return r.text
                  return None
              except Exception as e:
                  print(f"[!] Ошибка загрузки {url}: {e}")
                  return None

          def detect_from_text(text, flags):
              if not text:
                  return flags
              low = text.lower()
              if any(x in low for x in ANDROID_KEYS):
                  flags["android"] = True
              if any(x in low for x in IOS_KEYS):
                  flags["ios"] = True
              if any(x in low for x in WINDOWS_KEYS):
                  flags["windows"] = True
              if any(x in low for x in LINUX_KEYS):
                  flags["linux"] = True
              return flags

          def detect_from_asset_name(name):
              flags = {
                  "android": False,
                  "ios": False,
                  "windows": False,
                  "linux": False,
              }

              if not name:
                  return flags

              n = name.lower()

              if n.endswith((".apk", ".aab")) or "android" in n:
                  flags["android"] = True

              if n.endswith(".ipa") or "ios" in n or "iphone" in n or "ipad" in n:
                  flags["ios"] = True

              if (
                  n.endswith((".exe", ".msi", ".msix", ".appx"))
                  or "windows" in n
                  or ("win" in n and ("64" in n or "32" in n or "arm" in n))
              ):
                  flags["windows"] = True

              if (
                  n.endswith((".appimage", ".deb", ".rpm", ".snap", ".flatpak", ".pkg.tar.zst"))
                  or "linux" in n
                  or ("linux" in n and ".tar" in n)
              ):
                  flags["linux"] = True

              return flags

          def merge_flags(target, source):
              for key in target:
                  if source.get(key):
                      target[key] = True
              return target

          def platform_labels(flags):
              labels = []
              if flags.get("android"):
                  labels.append("Android")
              if flags.get("ios"):
                  labels.append("iOS")
              if flags.get("windows"):
                  labels.append("Windows")
              if flags.get("linux"):
                  labels.append("Linux")
              return labels

          def make_status(release_age_days, pushed_at):
              if release_age_days is None:
                  pushed_days = days_since(parse_date(pushed_at))
                  if pushed_days is not None and pushed_days <= 90:
                      return "no_release_recent_repo"
                  return "no_release"

              if release_age_days <= 365:
                  return "active"
              if release_age_days <= 730:
                  return "stale"
              return "outdated"

          def calculate_score(stars, flags, release_age_days, has_release):
              platforms_count = sum(1 for x in flags.values() if x)
              score = stars
              score += platforms_count * 2500
              if has_release:
                  if release_age_days is not None:
                      score += max(0, 6000 - release_age_days * 8)
              return int(score)

          def load_candidates():
              path = "raw/utils_candidates.jsonl"
              candidates = []

              if not os.path.exists(path):
                  print(f"[!] Файл не найден: {path}")
                  return candidates

              with open(path, "r", encoding="utf-8") as f:
                  for line in f:
                      line = line.strip()
                      if not line:
                          continue
                      try:
                          candidates.append(json.loads(line))
                      except Exception:
                          continue

              return candidates

          candidates = load_candidates()
          print(f"[*] Загружено кандидатов: {len(candidates)}")

          # Фильтруем по минимальным звёздам
          if MIN_STARS > 0:
              candidates = [c for c in candidates if c.get("stars", 0) >= MIN_STARS]
              print(f"[*] После фильтра по звёздам (>= {MIN_STARS}): {len(candidates)}")

          candidates.sort(key=lambda x: x.get("stars", 0), reverse=True)

          utilities = []

          for idx, candidate in enumerate(candidates[:MAX_PROBE], start=1):
              repo = candidate.get("repo")

              if not repo:
                  continue

              print(f"[*] Проверяю {idx}/{min(len(candidates), MAX_PROBE)}: {repo}")

              repo_api = f"{GITHUB_API}/repos/{repo}"
              repo_data = fetch_json(repo_api)

              if not repo_data:
                  print(f"[!] Не могу получить {repo}")
                  time.sleep(1)
                  continue

              full_name = repo_data.get("full_name", repo)
              html_url = repo_data.get("html_url", candidate.get("url"))
              name = repo_data.get("name", candidate.get("name", full_name.split("/")[-1]))
              description = repo_data.get("description") or candidate.get("description")
              language = repo_data.get("language")
              topics = repo_data.get("topics") or candidate.get("topics") or []
              stars = repo_data.get("stargazers_count", candidate.get("stars", 0))
              forks = repo_data.get("forks_count", candidate.get("forks", 0))
              pushed_at = repo_data.get("pushed_at")
              archived = repo_data.get("archived", False)

              flags = {
                  "android": False,
                  "ios": False,
                  "windows": False,
                  "linux": False,
              }

              # 1. Topics
              detect_from_text(" ".join(topics), flags)

              # 2. Name/description
              detect_from_text(name, flags)
              detect_from_text(description, flags)

              # 3. Releases
              release = fetch_json(f"{repo_api}/releases/latest")

              if not release:
                  releases = fetch_json(f"{repo_api}/releases", params={"per_page": 1})
                  if isinstance(releases, list) and releases:
                      release = releases[0]

              has_release = False
              latest_release_tag = None
              latest_release_name = None
              latest_release_url = None
              latest_release_published_at = None
              release_age_days = None
              release_assets = []

              if release:
                  has_release = True
                  latest_release_tag = release.get("tag_name")
                  latest_release_name = release.get("name") or release.get("tag_name")
                  latest_release_url = release.get("html_url")
                  latest_release_published_at = release.get("published_at") or release.get("created_at")
                  release_age_days = days_since(parse_date(latest_release_published_at))

                  assets = release.get("assets", []) or []

                  for asset in assets[:20]:
                      asset_name = asset.get("name")
                      asset_url = asset.get("browser_download_url")

                      if not asset_name or not asset_url:
                          continue

                      asset_flags = detect_from_asset_name(asset_name)
                      merge_flags(flags, asset_flags)

                      release_assets.append({
                          "name": asset_name,
                          "url": asset_url,
                          "size_mb": round((asset.get("size") or 0) / (1024 * 1024), 2),
                          "content_type": asset.get("content_type"),
                          "platform_hint": platform_labels(asset_flags),
                      })

              # 4. README, если платформ пока нет или включено принудительное чтение
              if INCLUDE_README and (not any(flags.values()) or not has_release):
                  readme_text = fetch_text(f"{repo_api}/readme", accept="application/vnd.github.raw")
                  if readme_text:
                      detect_from_text(readme_text[:80000], flags)

              labels = platform_labels(flags)

              status = make_status(release_age_days, pushed_at)

              score = calculate_score(
                  stars=stars,
                  flags=flags,
                  release_age_days=release_age_days,
                  has_release=has_release,
              )

              utilities.append({
                  "id": idx,
                  "name": name,
                  "full_name": full_name,
                  "url": html_url,
                  "description": description,
                  "language": language,
                  "topics": topics,
                  "stars": stars,
                  "forks": forks,
                  "archived": archived,
                  "pushed_at": pushed_at,
                  "query_slugs": candidate.get("query_slugs", []),
                  "has_release": has_release,
                  "latest_release_tag": latest_release_tag,
                  "latest_release_name": latest_release_name,
                  "latest_release_url": latest_release_url,
                  "latest_release_published_at": latest_release_published_at,
                  "release_age_days": release_age_days,
                  "platforms": flags,
                  "platform_labels": labels,
                  "platforms_text": ", ".join(labels) if labels else "Unknown",
                  "release_assets_count": len(release_assets),
                  "release_assets": release_assets[:12],
                  "status": status,
                  "score": score,
              })

              time.sleep(random.uniform(0.35, 1.1))

          # Финальная сортировка: сначала самые ценные
          utilities.sort(key=lambda x: x["score"], reverse=True)

          # Пересчитываем ID после сортировки
          for idx, item in enumerate(utilities, start=1):
              item["id"] = idx

          summary = {
              "total_utilities": len(utilities),
              "with_release": sum(1 for x in utilities if x["has_release"]),
              "without_release": sum(1 for x in utilities if not x["has_release"]),
              "android": sum(1 for x in utilities if x["platforms"]["android"]),
              "ios": sum(1 for x in utilities if x["platforms"]["ios"]),
              "windows": sum(1 for x in utilities if x["platforms"]["windows"]),
              "linux": sum(1 for x in utilities if x["platforms"]["linux"]),
              "active": sum(1 for x in utilities if x["status"] == "active"),
              "stale": sum(1 for x in utilities if x["status"] == "stale"),
              "outdated": sum(1 for x in utilities if x["status"] == "outdated"),
              "no_release": sum(1 for x in utilities if x["status"] == "no_release"),
              "no_release_recent_repo": sum(1 for x in utilities if x["status"] == "no_release_recent_repo"),
          }

          payload = {
              "generated_at": now_iso(),
              "generator": "utils-discovery.yml",
              "project": "REMAININGCONNECTIONS dashboard",
              "telegram_channel": "https://t.me/REMAININGCONNECTIONS",
              "candidates_checked": len(utilities),
              "summary": summary,
              "utilities": utilities,
          }

          os.makedirs(os.path.dirname(UTILS_OUTPUT), exist_ok=True)

          with open("probed/utils_probed.json", "w", encoding="utf-8") as f:
              json.dump(payload, f, ensure_ascii=False, indent=2)

          with open(UTILS_OUTPUT, "w", encoding="utf-8") as f:
              json.dump(payload, f, ensure_ascii=False, indent=2)

          print("==================================================")
          print(" Поиск утилит завершён")
          print("==================================================")
          print(f"Утилит: {summary['total_utilities']}")
          print(f"С релизом: {summary['with_release']}")
          print(f"Android: {summary['android']}")
          print(f"iOS: {summary['ios']}")
          print(f"Windows: {summary['windows']}")
          print(f"Linux: {summary['linux']}")
          print(f"Сохранено в {UTILS_OUTPUT}")
          print("==================================================")
          PY

            function escapeHtml(value) {
                if (value === null || value === undefined) {
                    return "";
                }

                return String(value)
                    .replaceAll("&", "&amp;")
                    .replaceAll("<", "&lt;")
                    .replaceAll(">", "&gt;")
                    .replaceAll('"', "&quot;")
                    .replaceAll("'", "&#039;");
            }

            function formatNumber(value) {
                const n = Number(value || 0);

                if (n >= 1_000_000) {
                    return (n / 1_000_000).toFixed(1).replace(/\.0$/, "") + "M";
                }

                if (n >= 1_000) {
                    return (n / 1_000).toFixed(1).replace(/\.0$/, "") + "k";
                }

                return String(n);
            }

            function formatMinutes(mins) {
                const m = Number(mins);

                if (!Number.isFinite(m)) {
                    return "unknown";
                }

                if (m < 1) {
                    return "сейчас";
                }

                if (m < 60) {
                    return `${Math.floor(m)} мин`;
                }

                if (m < 1440) {
                    return `${Math.floor(m / 60)} ч`;
                }

                if (m < 525600) {
                    return `${Math.floor(m / 1440)} дн`;
                }

                return `${Math.floor(m / 525600)} г`;
            }

            function getFilteredSubscriptions() {
                let items = [...state.subscriptions];

                const q = state.search.trim().toLowerCase();

                if (q) {
                    items = items.filter(item => {
                        return (
                            String(item.name || "").toLowerCase().includes(q) ||
                            String(item.url || "").toLowerCase().includes(q) ||
                            String(item.subscription_url || "").toLowerCase().includes(q) ||
                            String(item.last_commit_message || "").toLowerCase().includes(q)
                        );
                    });
                }

                if (state.source !== "all") {
                    items = items.filter(item => item.source === state.source);
                }

                if (state.status === "bs") {
                    items = items.filter(item => item.has_bs);
                }

                if (state.status === "cs") {
                    items = items.filter(item => !item.has_bs);
                }

                if (["active", "stale", "unknown", "dead", "error"].includes(state.status)) {
                    items = items.filter(item => item.status === state.status);
                }

                if (state.sort === "updated") {
                    items.sort((a, b) => {
                        const aa = a.updated_mins_ago ?? 999999999;
                        const bb = b.updated_mins_ago ?? 999999999;
                        return aa - bb;
                    });
                }

                if (state.sort === "configs") {
                    items.sort((a, b) => (b.configs_count || 0) - (a.configs_count || 0));
                }

                if (state.sort === "name") {
                    items.sort((a, b) => String(a.name).localeCompare(String(b.name), "ru"));
                }

                return items;
            }

                        );

                if (state.status === "online") {
                    items = items.filter(item => item.status === "online");
                }

                if (state.status === "offline") {
                    items = items.filter(item => item.status !== "online");

                if (state.sort === "ping") {
                    items.sort((a, b) => {
                        const aa = a.ping ?? 999999;
                        const bb = b.ping ?? 999999;

                    });
                } else {
                    items.sort((a, b) => {
                        const onlineA = a.status === "online" ? 0 : 1;
                        const onlineB = b.status === "online" ? 0 : 1;

                        if (onlineA !== onlineB) {
                            return onlineA - onlineB;
                        }

                        const aa = a.ping ?? 999999;
                        const bb = b.ping ?? 999999;

                        return aa - bb;
                    });
                }

                return items;
            }

# 2. Утилиты

def collapse_spaces(value: str) -> str:
    return re.sub(r"\s+", " ", value or "").strip()


def make_query_id(query: str) -> str:
    return hashlib.md5(query.encode("utf-8")).hexdigest()[:12]


def normalize_for_dedupe(query: str) -> str:
    return collapse_spaces(query.lower())


def build_query(*parts) -> str:
    cleaned = []

    for part in parts:
        if not part:
            continue

        if isinstance(part, (list, tuple, set)):
            part = " ".join(str(x) for x in part if x)

        part = collapse_spaces(str(part))

        if part:
            cleaned.append(part)

    return collapse_spaces(" ".join(cleaned))


def make_entry(category: str, qtype: str, query: str, priority: int, reason: str, qualifiers=None) -> dict:
    query = collapse_spaces(query)

    if not query:
        return None

    qualifiers = qualifiers or []

    full_query_parts = [query]

    if qualifiers:
        full_query_parts.extend(qualifiers)

    full_query = collapse_spaces(" ".join(full_query_parts))

    return {
        "id": make_query_id(full_query),
        "category": category,
        "type": qtype,
        "query": query,
        "full_query": full_query,
        "qualifiers": qualifiers,
        "priority": int(priority),
        "reason": reason,
    }


def finalize_entries(entries, limit: int, category: str, qtype: str):
    seen = set()
    unique = []

    for entry in entries:
        if not entry:
            continue

        key = normalize_for_dedupe(entry["full_query"])

        if key in seen:
            continue

        seen.add(key)
        unique.append(entry)

    # Небольшой шум для разнообразия при одинаковом приоритете
    for entry in unique:
        entry["_noise"] = random.random()

    unique.sort(
        key=lambda x: (
            x["priority"],
            x["_noise"],
        ),
        reverse=True,
    )

    if limit > 0 and len(unique) > limit:
        unique = unique[:limit]

    result = []

    for idx, entry in enumerate(unique, start=1):
        entry.pop("_noise", None)
        entry["slug"] = f"{category}-{qtype}-{idx:04d}"
        result.append(entry)

    return result

# 3. Автоадаптация по уже найденным данным
# =========================================================

def harvest_value(value, counter: Counter):
    if value is None:
        return

    if isinstance(value, str):
        words = WORD_RE.findall(value.lower())

        for word in words:
            word = word.strip()

            if len(word) < 3:
                continue

            if word in STOPWORDS:
                continue

            counter[word] += 1

    elif isinstance(value, (list, tuple, set)):
        for item in value:
            harvest_value(item, counter)


def harvest_terms_from_json(obj, counter: Counter):
    if isinstance(obj, dict):
        for key, value in obj.items():
            if key in HARVEST_KEYS:
                harvest_value(value, counter)
            else:
                harvest_terms_from_json(value, counter)

    elif isinstance(obj, list):
        for item in obj:
            harvest_terms_from_json(item, counter)


def load_adaptive_terms(paths, top_n: int = 80):
    counter = Counter()

    for path in paths:
        if not path:
            continue

        if not os.path.exists(path):
            continue

        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)

            harvest_terms_from_json(data, counter)
            print(f"[+] Harvested terms from: {path}")
        except Exception as e:
            print(f"[!] Failed to harvest {path}: {e}")

    terms = [word for word, _ in counter.most_common(top_n)]
    print(f"[+] Adaptive terms loaded: {len(terms)}")

    return terms

def generate_target_queries(target: str, aliases=None):
    entries = []

    if not target:
        return entries

    aliases = aliases or []

    variants = set()

    base = collapse_spaces(target)

    if base:
        variants.add(base)
        variants.add(base.lower())
        variants.add(base.upper())
        variants.add(base.replace(" ", "-"))
        variants.add(base.replace(" ", "_"))
        variants.add(base.replace(" ", ""))
        variants.add(base.replace("-", " "))
        variants.add(base.replace("_", " "))

    for alias in aliases:
        alias = collapse_spaces(alias)

        if not alias:
            continue

        variants.add(alias)
        variants.add(alias.lower())
        variants.add(alias.replace(" ", "-"))
        variants.add(alias.replace(" ", "_"))
        variants.add(alias.replace(" ", ""))

    variants = sorted(variants)

    for variant in variants:
        entries.append(
            make_entry(
                category="target",
                qtype="repo",
                query=f'"{variant}" in:name',
                priority=100,
                reason="target exact name",
                qualifiers=["fork:true", "archived:false"],
            )
        )

        entries.append(
            make_entry(
                category="target",
                qtype="repo",
                query=f'"{variant}" in:name,description,readme',
                priority=98,
                reason="target name/description/readme",
                qualifiers=["fork:true", "archived:false"],
            )
        )

        entries.append(
            make_entry(
                category="target",
                qtype="repo",
                query=f'"{variant}" mirror',
                priority=95,
                reason="target mirror",
                qualifiers=["fork:true", "archived:false"],
            )
        )

        entries.append(
            make_entry(
                category="target",
                qtype="repo",
                query=f'"{variant}" fork',
                priority=90,
                reason="target fork",
                qualifiers=["fork:true", "archived:false"],
            )
        )

        entries.append(
            make_entry(
                category="target",
                qtype="repo",
                query=f'"{variant}" alternative',
                priority=85,
                reason="target alternative",
                qualifiers=["fork:true", "archived:false"],
            )
        )

        entries.append(
            make_entry(
                category="target",
                qtype="repo",
                query=f'"{variant}" release',
                priority=80,
                reason="target release",
                qualifiers=["fork:true", "archived:false"],
            )
        )

        entries.append(
            make_entry(
                category="target",
                qtype="code",
                query=f'"{variant}"',
                priority=92,
                reason="target code mention",
            )
        )

    # Если target указан как owner/repo
    if "/" in base:
        owner, repo = base.split("/", 1)

        entries.append(
            make_entry(
                category="target",
                qtype="repo",
                query=f"user:{owner}",
                priority=99,
                reason="target owner",
                qualifiers=["fork:true", "archived:false"],
            )
        )

        entries.append(
            make_entry(
                category="target",
                qtype="repo",
                query=f"{owner} {repo}",
                priority=99,
                reason="target owner repo",
                qualifiers=["fork:true", "archived:false"],
            )
        )

    return finalize_entries(entries, limit=500, category="target", qtype="mixed")

def write_json(path: str, payload: dict):
    os.makedirs(os.path.dirname(path), exist_ok=True)

    with open(path, "w", encoding="utf-8") as f:
        json.dump(payload, f, ensure_ascii=False, indent=2)

    print(f"[+] Saved JSON: {path}")


def write_txt(path: str, entries):
    os.makedirs(os.path.dirname(path), exist_ok=True)

    with open(path, "w", encoding="utf-8") as f:
        for entry in entries:
            f.write(entry["full_query"] + "\n")

    print(f"[+] Saved TXT: {path}")


def write_yaml(path: str, entries):
    os.makedirs(os.path.dirname(path), exist_ok=True)

    lines = [
        "# Auto-generated search queries",
        "# Do not edit manually if generated by CI.",
        "queries:",
    ]

    for entry in entries:
        lines.append(f"  - slug: {json.dumps(entry['slug'], ensure_ascii=False)}")
        lines.append(f"    query: {json.dumps(entry['query'], ensure_ascii=False)}")
        lines.append(f"    full_query: {json.dumps(entry['full_query'], ensure_ascii=False)}")
        lines.append(f"    category: {json.dumps(entry['category'], ensure_ascii=False)}")
        lines.append(f"    type: {json.dumps(entry['type'], ensure_ascii=False)}")
        lines.append(f"    priority: {entry['priority']}")
        lines.append(f"    reason: {json.dumps(entry['reason'], ensure_ascii=False)}")

    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(lines) + "\n")

    print(f"[+] Saved YAML: {path}")


def write_matrix_json(path: str, entries):

def generate_all(args):
    random.seed(args.seed)

    adaptive_terms = load_adaptive_terms(args.existing_data, top_n=args.adaptive_terms)

    target_terms = []

    if args.target:
        target_terms = [args.target]

        if args.target_aliases:
            target_terms.extend(args.target_aliases)

    # TARGET
    target_entries = generate_target_queries(args.target, args.target_aliases or [])

    # SUBSCRIPTIONS
    sub_repo_raw = generate_subscription_repo_queries(adaptive_terms, target_terms)
    sub_code_raw = generate_subscription_code_queries(adaptive_terms)
    sub_topic_raw = generate_subscription_topic_queries()
    sub_gitverse_raw = generate_subscription_gitverse_queries(adaptive_terms)

    sub_repo = finalize_entries(sub_repo_raw, args.max_repo_queries, "subscriptions", "repo")
    sub_code = finalize_entries(sub_code_raw, args.max_code_queries, "subscriptions", "code")
    sub_topic = finalize_entries(sub_topic_raw, args.max_topic_queries, "subscriptions", "topic")
    sub_gitverse = finalize_entries(sub_gitverse_raw, args.max_gitverse_queries, "subscriptions", "gitverse")

    # TG PROXIES
    tg_repo_raw = generate_tg_proxy_repo_queries(adaptive_terms, target_terms)
    tg_code_raw = generate_tg_proxy_code_queries(adaptive_terms)
    tg_gitverse_raw = generate_tg_proxy_gitverse_queries(adaptive_terms)

    tg_repo = finalize_entries(tg_repo_raw, args.max_repo_queries, "tg_proxies", "repo")
    tg_code = finalize_entries(tg_code_raw, args.max_code_queries, "tg_proxies", "code")
    tg_gitverse = finalize_entries(tg_gitverse_raw, args.max_gitverse_queries, "tg_proxies", "gitverse")

    # UTILITIES
    util_repo_raw = generate_utilities_repo_queries(adaptive_terms, target_terms)
    util_code_raw = generate_utilities_code_queries(adaptive_terms)
    util_topic_raw = generate_utilities_topic_queries()

def iso_now():
    return (
        datetime.datetime.now(datetime.timezone.utc)
        .isoformat()
        .replace("+00:00", "Z")
    )


def iso_minutes_ago(minutes):
    dt = datetime.datetime.now(datetime.timezone.utc) - datetime.timedelta(minutes=minutes)
    return dt.isoformat().replace("+00:00", "Z")

def slugify(value):

def make_content_sample(protocol, path, region):
    ip = make_fake_ip()
    port = random.choice([443, 8443, 2053, 2083, 2087, 2096, 8080, 8880])
    node_name = f"{region}-{random.randint(100, 999)}"

    if path and path.lower().endswith((".yaml", ".yml")):
        return "\n".join(
            [
                "proxies:",
                f"  - name: '{node_name}'",
                f"    type: {protocol if protocol != 'wireguard' else 'wireguard'}",
                f"    server: {ip}",
                f"    port: {port}",
                "    udp: true",
                "    tls: true",
                f"    # generated_sample_for_{node_name}",
            ]
        )

    if protocol == "wireguard":
        return "\n".join(

                "[Interface]",
                f"PrivateKey = {node_name}FAKEKEY{random.randint(1000, 9999)}",
                f"Address = 10.0.0.{random.randint(2, 254)}/32",

                "",
                "[Peer]",
                f"PublicKey = {node_name}FAKEPUBLICKEY{random.randint(1000, 9999)}",
                f"Endpoint = {ip}:{port}",
                "AllowedIPs = 0.0.0.0/0, ::/0",
            ]
        )

    if protocol == "vmess":
        return f"vmess://eyJ2IjoiMiIsInBzIjoibm9kZS17node_name}",
        # Оставим простой читаемый вариант, чтобы не ломать восприятие:
        return f"vmess://{node_name}@{ip}:{port}?network=tcp&security=tls"

    if protocol == "shadowsocks":
        return f"ss://cmM0LW1kNTpwYXNzd29yZA==@{ip}:{port}#{node_name}"

    if protocol == "trojan":
        return f"trojan://{node_name}@{ip}:{port}?security=tls&type=tcp#{node_name}"

    if protocol == "hysteria2":
        return f"hysteria2://{node_name}@{ip}:{port}?insecure=1#{node_name}"

    if protocol == "tuic":
        return f"tuic://{node_name}@{ip}:{port}?congestion_control=bbr#{node_name}"

    return f"{protocol}://{node_name}@{ip}:{port}?security=tls&type=tcp#{node_name}"

def generate_subscription(idx):
    # Источники: примерно каждый 7-й будет Gitverse
    source = "gitverse" if idx % 7 == 0 else "github"

    protocol = random.choice(PROTOCOLS)
    region = random.choice(REGIONS)
    core = random.choice(NAME_CORES)

    name = slugify(f"{core}-{protocol}-{region}-{idx:03d}")
    org = random.choice(GITVERSE_ORGS if source == "gitverse" else GITHUB_ORGS)
    repo_full = f"{org}/{name}"

    branch = random.choice(["main", "master"])

    # Обновление: свежие / средние / старые
    r = random.random()
    if r < 0.52:
        updated_mins_ago = random.randint(1, 240)
    elif r < 0.85:
        updated_mins_ago = random.randint(241, 4320)
    else:
        updated_mins_ago = random.randint(4321, 120000)

    # Есть ли конфиги
    has_configs = random.random() > 0.075
    configs_count = 0 if not has_configs else random.randint(6, 9800)

    # Путь к файлу подписки
    has_path = random.random() > 0.08
    config_path = random.choice(CONFIG_PATHS) if has_path else None

    # URL
    if source == "github":
        repo_url = f"https://github.com/{repo_full}"
    else:
        repo_url = f"https://gitverse.ru/{repo_full}"

    # subscription_url
    subscription_url = None
    if has_configs and config_path:
        if source == "github":
            subscription_url = f"https://raw.githubusercontent.com/{repo_full}/{branch}/{config_path}"
        else:
            subscription_url = f"https://gitverse.ru/{repo_full}/raw/branch/{branch}/{config_path}"

    # BS
    has_bs = False
    if has_configs:
        has_bs = random.random() < 0.44

    # Статус
    if configs_count <= 0:
        status = "unknown"
    elif updated_mins_ago > 20160:
        status = "stale"
    else:
        status = "active"

    # content_sample
    content_sample = None
    if has_configs:
        content_sample = make_content_sample(
            protocol=protocol,
            path=config_path or "sub.txt",
            region=region,
        )

    return {
        "id": idx,
        "name": name,
        "source": source,
        "source_label": "GitHub" if source == "github" else "Gitverse",
        "repo": repo_full,
        "url": repo_url,
        "subscription_url": subscription_url,
        "config_path": config_path,
        "protocol_hint": protocol,
        "region_hint": region,
        "updated_mins_ago": updated_mins_ago,
        "updated_iso": iso_minutes_ago(updated_mins_ago),
        "configs_count": configs_count,
        "has_bs": has_bs,
        "bs_label": "БС" if has_bs else "ЧС",
        "status": status,
        "health_score": random.randint(10, 100) if has_configs else 0,
        "last_commit_message": random.choice(COMMIT_MESSAGES),
        "content_sample": content_sample,
        "tags": [
            source,
            protocol,
            region,
            status,
            "bs" if has_bs else "cs",
        ],
    }

def generate_payload(count):
    subscriptions = [
        generate_subscription(i)
        for i in range(1, count + 1)
    ]

    github_count = sum(1 for x in subscriptions if x["source"] == "github")
    gitverse_count = sum(1 for x in subscriptions if x["source"] == "gitverse")

    active_count = sum(1 for x in subscriptions if x["status"] == "active")
    stale_count = sum(1 for x in subscriptions if x["status"] == "stale")
    unknown_count = sum(1 for x in subscriptions if x["status"] == "unknown")

    bs_count = sum(1 for x in subscriptions if x["has_bs"])
    cs_count = len(subscriptions) - bs_count

    total_configs = sum(x["configs_count"] for x in subscriptions)

    top_by_configs = sorted(
        subscriptions,
        key=lambda x: x["configs_count"],
        reverse=True,
    )[:25]

    top_by_freshness = sorted(
        subscriptions,
        key=lambda x: x["updated_mins_ago"],
    )[:25]

    return {
        "generated_at": iso_now(),
        "generator": "generate_found.py",
        "schema_version": "1.0",
        "project": "REMAININGCONNECTIONS dashboard",
        "telegram_channel": "https://t.me/REMAININGCONNECTIONS",
        "candidates_found": len(subscriptions) * 17 + random.randint(111, 999),
        "probed_count": len(subscriptions),
        "summary": {
            "total_subscriptions": len(subscriptions),
            "github": github_count,
            "gitverse": gitverse_count,
            "active": active_count,
            "stale": stale_count,
