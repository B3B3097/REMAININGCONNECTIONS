function normalizeSubscriptions(payload) { let items = []; if (Array.isArray(payload)) { items = payload; } else if (payload && Array.isArray(payload.subscriptions)) { items = payload.subscriptions; } else if (payload && Array.isArray(payload.items)) { items = payload.items; } return items .map((item, index) => { const source = String(item.source || "github").toLowerCase(); return { id: item.id || index + 1, name: item.name || item.repo || "Unknown", source, source_label: source === "gitverse" ? "Gitverse" : "GitHub", url: item.url || item.subscription_url || "#", subscription_url: item.subscription_url || item.url || "", updated_mins_ago: item.updated_mins_ago ?? item.updated_minutes_ago ?? null, configs_count: Number(item.configs_count ?? item.configs ?? 0), has_bs: Boolean(item.has_bs ?? item.bs ?? false), status: item.status || "unknown", last_commit_message: item.last_commit_message || item.last_commit || "" }; }) .filter(item => item.configs_count > 0 && item.subscription_url && item.subscription_url !== "#"); }
